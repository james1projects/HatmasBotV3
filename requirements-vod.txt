# Kept out of requirements.txt on purpose: these pull ~1.5 GB of CUDA
# runtime wheels, which CI and the bot itself never need. The public
# site only READS the index (sqlite3 + ffmpeg), so the bot runs without
# any of this. Install once on the streaming PC:
#
#     pip install -r requirements-vod.txt
#
# Model weights (~3 GB for large-v3) download from Hugging Face on the
# first `python tools\vod_index.py` run and are cached under
# %USERPROFILE%\.cache\huggingface.

nvidia-cudnn-cu12>=9.25         # (vodsearch/cuda.py puts them on PATH; see its docstring)
